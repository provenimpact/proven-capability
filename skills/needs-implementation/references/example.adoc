import { Given, When, Then } from "@cucumber/cucumber";
import assert from "node:assert";

// --- Steps used by @CART-001 and @CART-003 scenarios ---

Given("the following products exist:", async function (dataTable) {
  const products = dataTable.hashes();
  // Seed the test database with products
  await this.db.query("DELETE FROM products");
  for (const product of products) {
    await this.db.query(
      "INSERT INTO products (name, price) VALUES ($1, $2)",
      [product.name, parseFloat(product.price)]
    );
  }
});

When("I add {string} to the cart", async function (productName) {
  const response = await this.api.post("/api/cart/items", {
    productName,
  });
  this.lastResponse = response;
});

Then("the cart count should be {int}", async function (expectedCount) {
  const response = await this.api.get("/api/cart");
  assert.strictEqual(response.body.items.length, expectedCount);
});

Then("I should see a confirmation {string}", async function (message) {
  assert.strictEqual(this.lastResponse.body.message, message);
});

Given("{string} is in my cart with quantity {int}", async function (productName, quantity) {
  // Add product to cart via API, then verify quantity
  for (let i = 0; i < quantity; i++) {
    await this.api.post("/api/cart/items", { productName });
  }
});

When("I view my cart", async function () {
  this.lastResponse = await this.api.get("/api/cart");
});

Then("I should see {int} distinct items", async function (count) {
  assert.strictEqual(this.lastResponse.body.items.length, count);
});

Then("the cart total should be {float}", async function (expectedTotal) {
  assert.strictEqual(this.lastResponse.body.total, expectedTotal);
});

Read `data-model.adoc` for Cart and CartItem entity definitions. Implement:

* Create `lib/db/migrations/004_create_carts.sql`:
+
....
CREATE TABLE carts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  session_id VARCHAR(255) NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_carts_session_id ON carts(session_id);

CREATE TABLE cart_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  cart_id UUID NOT NULL REFERENCES carts(id) ON DELETE CASCADE,
  product_id UUID NOT NULL REFERENCES products(id),
  quantity INTEGER NOT NULL CHECK (quantity >= 1),
  UNIQUE(cart_id, product_id)
);

CREATE INDEX idx_cart_items_cart_id ON cart_items(cart_id);
....

* Create `lib/models/cart.ts` with TypeScript types and basic query functions
